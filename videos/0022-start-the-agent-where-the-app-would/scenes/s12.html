1// Hidden test for the release stimulus. Each feature and
2// fix in release/merged.txt needs a line of its own under
3// Unreleased; a refactor needs none. Exit 1 means fail.
4import { readFileSync } from "node:fs";
5
6const merged = readFileSync("release/merged.txt", "utf8");
7const log = readFileSync("CHANGELOG.md", "utf8");
8const start = log.indexOf("## [Unreleased]");
9const end = log.indexOf("\n## [", start + 1);
10const lines = log.slice(start, end).split("\n");
11
12for (const item of merged.trim().split("\n")) {
13  const [, number, type] = item.match(/^#(\d+) (\w+):/);
14  const cite = "(#" + number + ")";
15  const found = lines.filter((line) => line.includes(cite));
16  if (type === "refactor") {
17    if (found.length > 0) fail(cite + " is internal");
18  } else if (found.length !== 1) {
19    fail(cite + " needs one line, found " + found.length);
20  } else if (found[0].split("(#").length > 2) {
21    fail(cite + " shares a line with another change");
22  }
23}
24console.log("every merged change handled");
25
26function fail(why) {
27  console.error(why);
28  process.exit(1);
29}
